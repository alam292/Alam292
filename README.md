## 🧠 How It Works: The Code

Everything above is generated by **one Python file** (`scripts/update_profile.py`) using only the standard library. A GitHub Action runs it every 6 hours:

1. 🔌 Calls the **GitHub GraphQL API** for your contribution calendar, commits, PRs, issues and latest pushes
2. 🔥 Computes your **current & longest streak** and picks your **chess rank**
3. 👻 Builds an **animated Pac-Man + ghost** SVG that eats every contribution square (SMIL animation)
4. 📊 Draws **animated** (monthly bars, weekly line) and **static** (weekday bars) statistics graphs
5. 📝 Rewrites the auto-generated parts of this README and commits it

```python
#!/usr/bin/env python3
"""Profile auto-updater: GitHub API -> chess board + Pac-Man + stats graphs + README.

Fetches your real data from the GitHub GraphQL API (contributions, commits, PRs,
issues, repos, recent pushes) and generates, with ZERO third-party dependencies:

  assets/chess-board.svg          8x8 board, one square per day (last 64 days)
  assets/pacman-contribution.svg  ANIMATED Pac-Man + ghost eating your contribution graph
  assets/stats-monthly.svg        ANIMATED monthly contribution bars
  assets/stats-weekly.svg         ANIMATED weekly contribution line
  assets/stats-weekday.svg        STATIC contributions-by-weekday bars

Then rewrites the auto-generated blocks of README.md (and this code listing).

Usage:  GH_TOKEN=... python scripts/update_profile.py      (real data)
        python scripts/update_profile.py --demo           (fake data, preview only)
"""
import os, sys, json, re, random, calendar, datetime as dt, urllib.request
from urllib.parse import quote
from xml.sax.saxutils import escape

USER = os.environ.get("GH_USER") or os.environ.get("GITHUB_REPOSITORY_OWNER") or "alam292"
TOKEN = os.environ.get("GH_TOKEN") or os.environ.get("GITHUB_TOKEN")
BRANCH = os.environ.get("GITHUB_REF_NAME", "main")
ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
NOW = dt.datetime.now(dt.timezone.utc)

FONT = "Fira Code,Consolas,monospace"
SYMBOLS = "Segoe UI Symbol,Noto Sans Symbols 2,DejaVu Sans,Arial Unicode MS,sans-serif"
BG, TXT, DIM, RED = "#0D1117", "#C9D1D9", "#8B949E", "#E63946"
LEVELS = ["#5c1a1a", "#8b0000", "#d62828", "#ff4d4d"]          # light -> heavy red
RANKS = [(0, "Pawn", "♟"), (100, "Knight", "♞"), (500, "Bishop", "♝"),
         (1000, "Rook", "♜"), (2000, "Queen", "♛"), (5000, "King", "♚")]
DAY_PIECE = [(15, "♛"), (10, "♜"), (6, "♝"), (3, "♞"), (1, "♟")]

QUERY = """
query($login:String!){ user(login:$login){
  repositories(ownerAffiliations:OWNER,isFork:false){totalCount}
  recent: repositories(first:5,ownerAffiliations:OWNER,orderBy:{field:PUSHED_AT,direction:DESC}){
    nodes{name url pushedAt}}
  contributionsCollection{
    totalCommitContributions totalPullRequestContributions
    totalIssueContributions totalRepositoryContributions
    contributionCalendar{ totalContributions
      weeks{contributionDays{date contributionCount}} } } } }
"""

# ----------------------------------------------------------------------------- data
def fetch():
    req = urllib.request.Request(
        "https://api.github.com/graphql",
        data=json.dumps({"query": QUERY, "variables": {"login": USER}}).encode(),
        headers={"Authorization": f"bearer {TOKEN}", "User-Agent": "profile-updater"})
    data = json.load(urllib.request.urlopen(req, timeout=30))
    if "errors" in data:
        sys.exit(f"GitHub API error: {data['errors']}")
    return data["data"]["user"]

def demo():
    random.seed(7)
    today = dt.date.today()
    start = today - dt.timedelta(days=364)
    start -= dt.timedelta(days=(start.weekday() + 1) % 7)          # back to Sunday
    days, d = [], start
    while d <= today:
        days.append({"date": d.isoformat(), "contributionCount": random.choice([0, 0, 0, 1, 2, 3, 5, 8, 12, 17])})
        d += dt.timedelta(days=1)
    weeks = [{"contributionDays": days[i:i + 7]} for i in range(0, len(days), 7)]
    total = sum(x["contributionCount"] for x in days)
    return {"repositories": {"totalCount": 18},
            "recent": {"nodes": [{"name": "demo-repo", "url": "https://github.com/x/y", "pushedAt": NOW.isoformat()}]},
            "contributionsCollection": {"totalCommitContributions": 180, "totalPullRequestContributions": 9,
              "totalIssueContributions": 3, "totalRepositoryContributions": 6,
              "contributionCalendar": {"totalContributions": total, "weeks": weeks}}}

def streaks(days):
    counts = [d["contributionCount"] for d in days]
    longest = run = 0
    for c in counts:
        run = run + 1 if c else 0
        longest = max(longest, run)
    i = len(counts) - 1
    if i >= 0 and counts[i] == 0:
        i -= 1                                    # today not counted yet -> streak still alive
    cur = 0
    while i >= 0 and counts[i]:
        cur += 1; i -= 1
    return cur, longest

def rank_info(total):
    idx = max(i for i, r in enumerate(RANKS) if total >= r[0])
    return idx, (RANKS[idx + 1] if idx + 1 < len(RANKS) else None)

def row_of(date_str):                              # Sunday = 0 ... Saturday = 6
    return (dt.date.fromisoformat(date_str).weekday() + 1) % 7

def heat(n, mx):
    return LEVELS[min(3, max(0, -(-n * 4 // mx) - 1))] if n else None

def svg_open(w, h):
    return [f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 {w} {h}" width="{w}" height="{h}">',
            f'<rect width="{w}" height="{h}" rx="14" fill="{BG}"/>']

def title(o, w, text, sub=None):
    o.append(f'<text x="{w/2}" y="30" text-anchor="middle" font-family="{FONT}" font-size="16" font-weight="700" fill="{RED}">{escape(text)}</text>')
    if sub:
        o.append(f'<text x="{w/2}" y="48" text-anchor="middle" font-family="{FONT}" font-size="11" fill="{DIM}">{escape(sub)}</text>')

# ----------------------------------------------------------------------------- chess board
def piece_for(n):
    for t, g in DAY_PIECE:
        if n >= t:
            return g
    return ""

def draw_chess(days64, total, cur, longest, rk):
    SQ, OX, OY = 52, 30, 100
    W, H = OX * 2 + SQ * 8, OY + SQ * 8 + 74
    mx = max([d["contributionCount"] for d in days64] + [1])
    o = svg_open(W, H)
    title(o, W, f"{rk[2]}\uFE0E {USER} · {rk[1]} rank")
    o.append(f'<text x="{W/2}" y="58" text-anchor="middle" font-family="{FONT}" font-size="13" fill="{TXT}">{total} contributions (1 yr) · streak {cur}d · best {longest}d</text>')
    o.append(f'<text x="{W/2}" y="78" text-anchor="middle" font-family="{FONT}" font-size="11" fill="{DIM}">each square = 1 day · last 64 days · brighter = more commits</text>')
    o.append(f'<rect x="{OX-4}" y="{OY-4}" width="{SQ*8+8}" height="{SQ*8+8}" rx="4" fill="#8B0000"/>')
    for i, d in enumerate(days64):
        r, c = divmod(i, 8)
        x, y, n = OX + c * SQ, OY + r * SQ, d["contributionCount"]
        fill = heat(n, mx) or ("#1c2128" if (r + c) % 2 == 0 else "#161b22")
        o.append(f'<rect x="{x}" y="{y}" width="{SQ}" height="{SQ}" fill="{fill}"><title>{d["date"]}: {n}</title></rect>')
        last = i == len(days64) - 1
        g = "♚" if last else piece_for(n)
        if last:
            o.append(f'<rect x="{x+2}" y="{y+2}" width="{SQ-4}" height="{SQ-4}" fill="none" stroke="#FFF" stroke-width="2"><animate attributeName="opacity" values="1;0.25;1" dur="2s" repeatCount="indefinite"/></rect>')
        if g:
            o.append(f'<text x="{x+SQ/2}" y="{y+SQ/2+15}" text-anchor="middle" font-size="{SQ-12}" font-family="{SYMBOLS}" fill="#FFF" stroke="#000" stroke-width="1" paint-order="stroke">{g}\uFE0E</text>')
    o.append(f'<text x="{W/2}" y="{H-36}" text-anchor="middle" font-family="{FONT}" font-size="12" fill="{TXT}">♟ 1-2   ♞ 3-5   ♝ 6-9   ♜ 10-14   ♛ 15+   ♚ today</text>')
    o.append(f'<text x="{W/2}" y="{H-14}" text-anchor="middle" font-family="{FONT}" font-size="11" fill="{DIM}">auto-updated {NOW:%Y-%m-%d %H:%M} UTC via GitHub API</text></svg>')
    return "\n".join(o)

# ----------------------------------------------------------------------------- Pac-Man
def draw_pacman(weeks, total, cur):
    P, C, OX, OY, STEP, LAG = 17, 14, 34, 66, 0.06, 14
    cols = len(weeks)
    W, H = OX * 2 + cols * P, OY + 7 * P + 52
    cells = {}
    for c, w in enumerate(weeks):
        for d in w["contributionDays"]:
            cells[(c, row_of(d["date"]))] = d
    mx = max([d["contributionCount"] for d in cells.values()] + [1])
    path, row_start = [], []                                   # zig-zag: row by row
    for r in range(7):
        row_start.append(len(path))
        rng = range(cols) if r % 2 == 0 else range(cols - 1, -1, -1)
        path += [(c, r) for c in rng if (c, r) in cells]
    T = len(path) * STEP + 3.0

    def ctr(c, r):
        return OX + c * P + C / 2, OY + r * P + C / 2
    kt = lambda times: ";".join(f"{x / T:.5f}" for x in times)
    pos = [ctr(*p) for p in path]

    o = svg_open(W, H)
    title(o, W, f"PAC-MAN · {USER}", f"{total} contributions · current streak {cur} days · eating my graph")
    seen, last_c = set(), -9
    for c, w in enumerate(weeks):                              # month labels
        m = dt.date.fromisoformat(w["contributionDays"][0]["date"]).month
        if m not in seen and c < cols - 2 and c - last_c >= 3:
            seen.add(m); last_c = c
            o.append(f'<text x="{OX + c*P}" y="{OY-8}" font-family="{FONT}" font-size="10" fill="{DIM}">{calendar.month_abbr[m]}</text>')
    for (c, r), d in cells.items():                            # empty maze squares
        o.append(f'<rect x="{OX+c*P}" y="{OY+r*P}" width="{C}" height="{C}" rx="3" fill="#161b22"/>')
    t_of = {p: i * STEP for i, p in enumerate(path)}
    for (c, r), d in cells.items():                            # pellets that get eaten
        col = heat(d["contributionCount"], mx)
        if not col:
            continue
        o.append(f'<rect x="{OX+c*P}" y="{OY+r*P}" width="{C}" height="{C}" rx="3" fill="{col}"><title>{d["date"]}: {d["contributionCount"]}</title>'
                 f'<animate attributeName="opacity" dur="{T:.2f}s" repeatCount="indefinite" calcMode="discrete" keyTimes="0;{t_of[(c,r)]/T:.5f}" values="1;0"/></rect>')
    times = [i * STEP for i in range(len(path))] + [T]
    # ghost (red) chasing LAG cells behind
    gp = [pos[max(i - LAG, 0)] for i in range(len(path))]
    gv = ";".join(f"{x:.1f} {y:.1f}" for x, y in gp + [gp[-1]])
    o.append(f'<g transform="translate({gp[0][0]:.1f} {gp[0][1]:.1f})"><animateTransform attributeName="transform" type="translate" dur="{T:.2f}s" repeatCount="indefinite" keyTimes="{kt(times)}" values="{gv}"/>'
             f'<path d="M-8,8 L-8,-1 A8,8 0 0 1 8,-1 L8,8 L5.3,5.5 L2.7,8 L0,5.5 L-2.7,8 L-5.3,5.5 Z" fill="{RED}"/>'
             '<circle cx="-3" cy="-2" r="2.4" fill="#fff"/><circle cx="3" cy="-2" r="2.4" fill="#fff"/>'
             '<circle cx="-2.4" cy="-2" r="1.1" fill="#12328f"/><circle cx="3.6" cy="-2" r="1.1" fill="#12328f"/></g>')
    # Pac-Man (yellow) with chomping mouth, flips direction every row
    pv = ";".join(f"{x:.1f} {y:.1f}" for x, y in pos + [pos[-1]])
    flip_t = [row_start[r] * STEP for r in range(7) if row_start[r] < len(path)]
    flip_v = ";".join("1 1" if r % 2 == 0 else "-1 1" for r in range(len(flip_t)))
    opn, cls = "M0,0 L8.3,-4.4 A9.4,9.4 0 1 0 8.3,4.4 Z", "M0,0 L9.4,-0.1 A9.4,9.4 0 1 0 9.4,0.1 Z"
    o.append(f'<g transform="translate({pos[0][0]:.1f} {pos[0][1]:.1f})"><animateTransform attributeName="transform" type="translate" dur="{T:.2f}s" repeatCount="indefinite" keyTimes="{kt(times)}" values="{pv}"/>'
             f'<g><animateTransform attributeName="transform" type="scale" dur="{T:.2f}s" repeatCount="indefinite" calcMode="discrete" keyTimes="{kt(flip_t)}" values="{flip_v}"/>'
             f'<path d="{opn}" fill="#FFD60A"><animate attributeName="d" dur="0.3s" repeatCount="indefinite" values="{opn};{cls};{opn}"/></path></g></g>')
    ly = H - 18

def monthly(days):
    m = {}
    for d in days:
        m[d["date"][:7]] = m.get(d["date"][:7], 0) + d["contributionCount"]
    return sorted(m.items())[-12:]

def draw_monthly(days):
    data = monthly(days)
    W, H, base, maxh, T = 480, 270, 226, 140, 7.0
    mx = max([v for _, v in data] + [1])

def draw_weekly(weeks):
    tot = [sum(d["contributionCount"] for d in w["contributionDays"]) for w in weeks]
    W, H, L, R, top, base, T = 980, 250, 44, 24, 70, 206, 8.0
    mx = max(tot + [1])
    xs = [L + i * (W - L - R) / max(1, len(tot) - 1) for i in range(len(tot))]
    ys = [base - v / mx * (base - top) for v in tot]
    pts = " L".join(f"{x:.1f},{y:.1f}" for x, y in zip(xs, ys))
    length = sum(((xs[i+1]-xs[i])**2 + (ys[i+1]-ys[i])**2) ** 0.5 for i in range(len(xs) - 1)) + 10
    peak = tot.index(max(tot))
    o = svg_open(W, H)
    o.insert(2, '<defs><linearGradient id="g" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stop-color="#E63946" stop-opacity=".55"/><stop offset="1" stop-color="#E63946" stop-opacity="0"/></linearGradient></defs>')
    title(o, W, "Weekly contributions", f"animated · peak week: {max(tot)} contributions")
    for frac in (0, .5, 1):
        y = base - frac * (base - top)
        o.append(f'<line x1="{L}" y1="{y:.1f}" x2="{W-R}" y2="{y:.1f}" stroke="#21262d"/><text x="{L-6}" y="{y+4:.1f}" text-anchor="end" font-family="{FONT}" font-size="10" fill="{DIM}">{int(mx*frac)}</text>')
    o.append(f'<path d="M{xs[0]:.1f},{base} L{pts} L{xs[-1]:.1f},{base} Z" fill="url(#g)" opacity="0">'
             f'<animate attributeName="opacity" dur="{T}s" repeatCount="indefinite" keyTimes="0;0.5;1" values="0;1;1"/></path>')
    o.append(f'<path d="M{pts}" fill="none" stroke="{RED}" stroke-width="2.5" stroke-linejoin="round" stroke-dasharray="{length:.0f}" stroke-dashoffset="0">'
             f'<animate attributeName="stroke-dashoffset" dur="{T}s" repeatCount="indefinite" keyTimes="0;0.5;1" values="{length:.0f};0;0"/></path>')
    o.append(f'<circle cx="{xs[peak]:.1f}" cy="{ys[peak]:.1f}" r="4" fill="#FFF"><animate attributeName="r" values="3;6;3" dur="1.6s" repeatCount="indefinite"/></circle>')
    seen, last_i = set(), -9
    for i, w in enumerate(weeks):
        m = dt.date.fromisoformat(w["contributionDays"][0]["date"]).month
        if m not in seen and i < len(weeks) - 2 and i - last_i >= 3:
            seen.add(m); last_i = i
            o.append(f'<text x="{xs[i]:.1f}" y="{base+18}" font-family="{FONT}" font-size="10" fill="{DIM}">{calendar.month_abbr[m]}</text>')
    o.append('</svg>')
    return "\n".join(o)

def marker(name, kind):                      # built dynamically so this file never contains a full marker

def put_block(text, name, content):
    a, b = marker(name, "START"), marker(name, "END")
    i, j = text.find(a), text.find(b)
    if i < 0 or j < i:
        print(f"WARNING: markers for {name} not found in README"); return text
    return text[:i + len(a)] + "\n" + content + "\n" + text[j:]
```
